option(FRNLCPP_STRICT_WARNINGS "Enable strict warnings for event dispatch regressions" ON)
option(FRNLCPP_SANITIZERS "Instrument event dispatch regressions with ASan and UBSan" OFF)
if(FRNLCPP_SANITIZERS AND NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  message(FATAL_ERROR "FRNLCPP_SANITIZERS requires Clang or GCC")
endif()
function(frnlcpp_configure_test target)
  target_link_libraries(${target} PRIVATE frnlcpp_event_dispatch)
  # Release builds define NDEBUG, which disables assert-based regressions.
  # Explicitly undefine it after configuration flags for every test target.
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
  if(FRNLCPP_STRICT_WARNINGS)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endif()
  if(FRNLCPP_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()

set_target_properties(frnlcpp_semantic PROPERTIES EXPORT_NAME semantic)
target_compile_features(frnlcpp_semantic INTERFACE cxx_std_17)
target_include_directories(frnlcpp_semantic INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
install(TARGETS frnlcpp_semantic EXPORT FrnlCPPTargets)
install(DIRECTORY include/frnl DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT FrnlCPPTargets
  FILE FrnlCPPTargets.cmake
  NAMESPACE FrnlCPP::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FrnlCPP)
install(FILES cmake/FrnlCPPConfig.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FrnlCPP)
