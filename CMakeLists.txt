cmake_minimum_required(VERSION 3.12)
project(FrnlCPP LANGUAGES CXX)
add_library(frnlcpp INTERFACE)
target_include_directories(frnlcpp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(frnlcpp INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
  add_executable(frnlcpp_component_scope_test tests/component_scope_test.cpp)
  target_link_libraries(frnlcpp_component_scope_test PRIVATE frnlcpp)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(frnlcpp_component_scope_test PRIVATE -Wall -Wextra -Werror -pedantic)
  endif()
  add_test(NAME frnlcpp_component_scope COMMAND frnlcpp_component_scope_test)
endif()
